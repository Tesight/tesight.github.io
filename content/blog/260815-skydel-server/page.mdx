# 建立连接：`RemoteSimulator.connect()`

```Python
def connect(
        self, ip: str = "localhost", id: int = 0, failIfApiVersionMismatch: bool = False
    ) -> None:
    if self.isConnected():
        raise Exception("Cannot connect. Already connected. Disconnect first.")

    port = 4820 + id
    if self.verbose:
        print("Connecting to %s on port %s..." % (ip, port))
    # client 连接 Skydel server
    self.client = ClientCmd(ip, port)

    # 1. 检查 API 版本
    self._serverApiVersion = self.client.getServerApiVersion()
    if ApiVersion != self._serverApiVersion:
        msg = (
            "Client Api Version ("
            + str(ApiVersion)
            + ") != Server Api Version ("
            + str(self._serverApiVersion)
            + ")"
        )
        if failIfApiVersionMismatch:
            raise Exception(msg)
        else:
            print("Warning: " + msg)

    # 2. 通过 _callCommand() 推送命令，获取 HIL 端口
    hilPort = self._callCommand(GetHilPort()).port()
    # 创建一个 HIL client
    self.hil = ClientHil(self.client.getAddress(), hilPort)
```

## 获取 API 版本：`getServerApiVersion()`

在 `getServerApiVersion()` 方法中，整个流程可以分为发送请求和等待响应两步。

### 发送请求

```Python
# _msgId2Packet()
message = struct.pack("<B", MsgId.ApiVersion) + struct.pack("<I", ApiVersion)

# _sendMessage()
packet = struct.pack("<H", len(message)) + message
self.sock.sendall(packet)
```

上述过程构造了一个 2 + 1 + 4 字节的请求包，由 socket 发送：

`msg_len: unsigned short | MsgId.ApiVersion: unsigned char | ApiVersion: unsigned int`

实际上，通过 API 与 Skydel Server 通信时，收发的数据包都遵循这个格式：

|长度头（`unsigned short`）|消息标识（`unsigned char`）|消息内容|
|---|---|---|
|消息标识+消息内容的数据长度|`0`：指令消息<br/>`1`：指令执行的结果消息<br/>`2`：API 查询消息|发送或接收的真正数据|

### 等待响应

Skydel API 提供了一个用于获取数据的 helper 方法：

```Python
def _getPacket(self, size: int) -> bytes:
    # 尝试从 socket 读取 size 大小数据（因网络等原因实际未必为 size 大小）
    packet = self.sock.recv(size)
    while len(packet) != size:
        # 通过循环补到 size 大小
        chunk = self.sock.recv(size - len(packet))
        # 如果没有额外数据可补了，就是断线了
        if len(chunk) == 0:
            raise Exception("Server closed connection.")
        packet = packet + chunk
    # 数据一定是 size 大小
    return packet
```

`_getPacket()` 的具体功能是从 socket 中读取精确 `size` 大小的数据包，若没有数据可读时就抛出异常（断线）。在等待 API 版本的响应过程中，也是按照 **长度头 + 消息标识 + 消息内容** 的格式来读取：

```Python
# 持续等待 socket 的响应
while 1 == 1:
    # _getPacketSize()
    msgSize = struct.unpack("<H", self._getPacket(2))[0]
    # _getPacketMsgId()
    msgId = struct.unpack("<B", self._getPacket(1))[0]

    # 如果消息标识是 API 查询，那么这条响应消息就是正确的响应，读取 API 版本
    if msgId == MsgId.ApiVersion:
        return struct.unpack("<I", self._getPacket(4))[0]
    else: 
        # 如果消息标识不匹配，不是目标消息，就丢弃这条消息
        # msgSize - 1 是指除消息标识以后的消息内容长度，消息标识已经读取
        self._getPacket(msgSize - 1)
```

正确获取 API 版本后，要求 Server 与 Client API 版本一致，否则也会抛出异常。

## 推送命令：_callCommand()

```Python
hilPort = self._callCommand(GetHilPort()).port()
self.hil = ClientHil(self.client.getAddress(), hilPort)
```

接着 RemoteSimulator 通过 `_callCommand()` 获取 HIL 端口，并开启了一个 HIL Client。`_callCommand()` 是非常重要的一个方法，所有 Skydel 指令都以类似的方式发送到 Skydel Server。

`_callCommand()` 由发送命令 `_postCommand()` 和等待结果 `_waitCommand()` 两个步骤构成：

```Python
def _callCommand(self, cmd, timestamp=None):
    self._postCommand(cmd, timestamp)
    return self._waitCommand(cmd)
```

### 发送命令

`_postCommand()` 方法判断了弃用指令，并按要求给指令添加时间戳，由 `sendCommand()` 实际发送出指令：

```Python
def _postCommand(self, cmd, timestamp=None):
    # 判断该指令是否弃用了
    deprecated = cmd.deprecated()
    if deprecated and (
        self.deprecatedMessageMode == DeprecatedMessageMode.ALL
        or (
            self.deprecatedMessageMode == DeprecatedMessageMode.LATCH
            and cmd.getName() not in self.latchDeprecated
        )
    ):
        print(f"Warning: {deprecated}")
        self.latchDeprecated.add(cmd.getName())
    # 如果发送指令时指定了时间戳，那么会给指令添加时间戳
    if timestamp != None:
        cmd.setTimestamp(timestamp)
    # 真正发送指令的方法
    self.client.sendCommand(cmd)
    return cmd
```

在 `senCommand()` 中，可以看到发送命令过程真正发送的数据：

```Python
def sendCommand(self, cmd: CommandBase) -> None:
    json_str = cmd.toJson()

    message = (self._msgId2Packet(MsgId.Command)) + (json_str + "\0").encode(
        "UTF-8"
    )
    self._sendMessage(message)
```

这个过程实际上是发送了 JSON 格式的指令，并将其封装为以下统一格式：

|长度头（`unsigned short`）|消息标识（`unsigned char`）|消息内容|
|---|---|---|
|消息标识+消息内容的数据长度|`0`：表明该消息为指令消息|JSON 格式的指令 + `0`停止符（以 UTF-8 编码）|

需要注意的是，在 `CommandBase` 类会通过指令名 `CmdName` 和 UUID `CmdUuid` 等数据完成实例化，这些数据也就是 JSON 中的实际数据。

### 等待响应

在使用 `_postCommand()` 发送指令后，就进入 `_waitCommand()` 的等待响应阶段：

```Python
def _waitCommand(self, cmd):
    result = self.client.waitCommand(cmd)
    self._handleException(result)
    return result

def waitCommand(self, cmd: CommandBase) -> Optional[CommandResult]:
    while 1 == 1:
        msgSize = self._getPacketSize()
        msgId = self._getPacketMsgId()

        if msgId == MsgId.Result:
            msg_json_len = struct.unpack("<i", self._getPacket(4))[0]
            msg_json = self._getPacket(msg_json_len)[:-1]
            result = createCommandResult(msg_json)
            # 要求响应的 UUID 与发送指令的 UUID 匹配
            if cmd.getUuid() == result.getRelatedCommand().getUuid():
                return result
        else:
            self._getPacket(msgSize - 1)
```

等待响应的过程与 `getServerApiVersion()` 等待响应的过程类似，不过其中多了对响应的 UUID 与发送指令的 UUID 匹配的要求。`_handleException()` 方法根据响应结果判断了 Skydel Server 的状态，决定是否要抛出异常。

### post() 与 call()

Skydel API 给用户提供了两种发送指令的方法，分别是 `post()` 和 `call()`。实际上两个命令都是在调用 `_postCommand()` 这个核心方法，而不同之处在于 `call()` 在调用 `postCommand()` 后会立即调用 `_waitCommand()`，其中的 `while 1== 1` 会导致程序阻塞，直至拿到指令的响应或是与 Server 断开连接。在调用 `post()` 后则需要手动调用 `wait()` 才能拿到指令的响应。

```Python
def post(self, cmd, timestamp=None):
        self._checkConnect()
        self._checkForbiddenPost(cmd)
        if self.verbose:
            print("Post " + cmd.toString())
        self._postCommand(cmd, timestamp)
        return cmd
        
def wait(self, cmd):
        self._checkConnect()
        if self.verbose:
            sys.stdout.write("Wait " + cmd.toString())
            sys.stdout.flush()
        result = self._waitCommand(cmd)
        if self.verbose:
            print(" => " + result.getMessage())
        return result
```

```Python
def call(self, cmd, timestamp=None):
        self._checkConnect()
        self._checkForbiddenCall(cmd)
        self._postCommand(cmd, timestamp)
        if self.verbose:
            sys.stdout.write("Call " + cmd.toString())
            sys.stdout.flush()

        result = self._waitCommand(cmd)
        if self.verbose:
            print(" => " + result.getMessage())
        return result
```

# ClientHil

在 `RemoteSimulator.connect()` 的最后，由指令获取到的 HIL 端口实例化了一个 `ClientHil` 类，打开了 HIL 客户端。`ClientHil` 同样继承了 `Client` 类，主要用于在 HIL 模式中向 Skydel Server 推送实时的坐标位置。

## 消息类型

`ClientHil` 定义了 7 种消息类型标识（`MsgId`）：

|消息类型标识|意义|
|---|---|
|`Hello`|向 Skydel Server 握手|
|`Bye`|向 Skydel Server 挥手|
|`VehicleInfo`|Skydel Server 发来的车辆信息|
|`PushEcef`|向 Skydel Server 推送车辆位置信息|
|`PushEcefNed`|向 Skydel Server 推送车辆位置 + 姿态信息|
|`PushEcefDynamics`|向 Skydel Server 推送车辆位置 + 运动信息|
|`PushEcefNedDynamics`|向 Skydel Server 推送车辆位置 + 姿态 + 运动信息|

## pollVehicleInfo()

该方法的主要功能是从 socket 中查询接收车辆信息数据，若有则解析并返回；若无数据则返回 `None`。socket 在接收数据时有超时时间，若 socket 中没有已到达的数据，socket 就会在超时时间内等待网络数据的到来（阻塞等待）。`pollVehicleInfo()` 的主要操作是将 socket 的超时时间设置为 0（*`self`*`.sock.settimeout(0)`），不论是否获取到数据都立刻返回。避免在监控车辆状态这样频繁检查数据、实时性要求高的场景下发生主线程的阻塞。

```Python
def pollVehicleInfo(self):
        _start = datetime.datetime.now()
        sent = None
        oldTimeout = self.sock.gettimeout()
        # 设置 socket 超时时间为 0，切换到非阻塞模式
        self.sock.settimeout(0)
        try:
            # 接收来自 socket 的数据，最多 255 字节
            result, addr = self.sock.recvfrom(255)
            # 获取消息类型标识（第 1 字节）
            id = result[0]
            # 可能是字符类型，就转为 ASCII
            if type(result[0]) != int:
                id = ord(result[0])
            # 获取车辆信息
            if id == MsgId.VehicleInfo:
                sent = VehicleInfo()
                (
                    sent.elapsedTime,    # 运行时间 
                    sent.ecef.x,         # ECEF x
                    sent.ecef.y,         # ECEF x
                    sent.ecef.z,         # ECEF z
                    sent.attitude.yaw,   # 偏航角
                    sent.attitude.pitch, # 俯仰角
                    sent.attitude.roll,  # 横滚角
                    sent.speed,          # 速度
                    sent.heading,        # 朝向
                    sent.odometer,       # 里程
                ) = struct.unpack("<Qddddddddd", result[1:])
        
        except socket.timeout:
            pass
        except socket.error as e:
            if e.args[0] != socket.EWOULDBLOCK:
                raise
                
        # 设置 socket 超时时间，切换回阻塞模式
        self.sock.settimeout(oldTimeout)
        
        return sent
```

`RemoteSimulator()` 中提供了操作 `ClientHil` 获取信息的方法：

```Python
# 向 Skydel Server 发送指令，告诉开始接收车辆信息
def beginVehicleInfo(self) -> CommandResult:
    return self.call(BeginVehicleInfo())

# 向 Skydel Server 发送指令，告诉停止接收车辆信息
def endVehicleInfo(self) -> CommandResult:
    return self.call(EndVehicleInfo())

# 调用 pollVehicleInfo() 查询是否有车辆信息，若有就缓存到 _last_vehicle_info 属性中
def hasVehicleInfo(self) -> bool:
    pos = self.hil.pollVehicleInfo()
    if pos is not None:
        self._last_vehicle_info = pos
        return True
    return False

# 反复查询（阻塞）直至 _last_vehicle_info 属性中有车辆信息，并取出返回
def nextVehicleInfo(self):
    while self._last_vehicle_info is None:
        self.hasVehicleInfo()
    sent = self._last_vehicle_info
    self._last_vehicle_info = None
    return sent

# 抛弃掉所有已经到达的数据，拿到最新的一条
def lastVehicleInfo(self):
    while self.hasVehicleInfo():
        pass
    return self.nextVehicleInfo()
```

`example_vehicle_info.py` 例程中展示了以上车辆数据方法的调用周期：

整体流程是：

1. `sim.beginVehicleInfo()` 令 Skydel Server 开始发送车辆信息
2. `sim.start()` 开始仿真
3. `sim.lastVehicleInfo()` 获取最新的车辆信息，可以显示或写入日志
4. `sim.stop()` 停止仿真
5. `sim.endVehicleInfo()` 令 Skydel Server 不再发送车辆信息
6. `sim.disconnect()` 与 Skydel Server 断开连接

## pushEcef() & pushEcefNed()

`ClientHil` 中的这两个方法都是用于向服务端推送车辆的位置信息，两个方法都比较简单，操作流程都是将信息包装好后直接发送。

在 `pushEcef()` 中，位置信息采用 ECEF 坐标，包装了以下格式的消息：
| 字段      | 类型                  | 含义                                              |
| ------- | ------------------- | ----------------------------------------------- |
| 消息标识    | `unsigned char`     | `PushEcef` / `PushEcefDynamics`（根据是否包含加速度等运动信息） |
| 运动信息类型  | `unsigned char`（若有） | 加加速度（Jerk）                                      |
| 运动信息类型  | `unsigned char`（若有） | 加速度                                             |
| 运动信息类型  | `unsigned char`（若有） | 速度                                              |
| 时间      | `double`            | 流逝时间                                            |
| 位置 ECEF | `double * 3`        | ECEF 坐标的 x、y、z                                  |
| 速度      | `double * 3`（若有）    | 速度的 x、y、z 分量                                    |
| 加速度     | `double * 3`（若有）    | 加速度的 x、y、z 分量                                   |
| 加加速度    | `double * 3`（若有）    | 加加速度的 x、y、z 分量                                  |
